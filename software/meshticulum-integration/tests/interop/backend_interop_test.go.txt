package rest

import (
 "bytes"
 "context"
 "crypto/sha256"
 "encoding/base64"
 "encoding/hex"
 "encoding/json"
 "net/http/httptest"
 "os"
 "os/exec"
 "testing"
 "time"

 "github.com/stretchr/testify/require"
)

func TestMeshticulumEncryptedBackendInterop(t *testing.T) {
 python, script := os.Getenv("MESHTICULUM_INTEROP_PYTHON"), os.Getenv("MESHTICULUM_INTEROP_SCRIPT")
 require.NotEmpty(t, python); require.NotEmpty(t, script)
 first, second := newWorkflowTestFixture(t), newWorkflowTestFixture(t)
 firstHTTP, secondHTTP := httptest.NewServer(first.server.Router()), httptest.NewServer(second.server.Router())
 defer firstHTTP.Close(); defer secondHTTP.Close()
 configuration, err := json.Marshal(map[string]string{
  "first_url": firstHTTP.URL, "second_url": secondHTTP.URL,
  "first_seed": hex.EncodeToString(first.keys["first"].Seed()),
  "second_seed": hex.EncodeToString(second.keys["second"].Seed()),
 })
 require.NoError(t, err)
 ctx, cancel := context.WithTimeout(t.Context(), 90*time.Second); defer cancel()
 child := exec.CommandContext(ctx, python, script); child.Stdin = bytes.NewReader(configuration)
 output, err := child.CombinedOutput(); require.NoError(t, err, string(output)); t.Log(string(output))
 scalar := sha256.Sum256([]byte("MESHTICULUM-NOISE-SECRET-CANARY"))
 secretEncodings := [][]byte{scalar[:], []byte(hex.EncodeToString(scalar[:])), []byte(base64.StdEncoding.EncodeToString(scalar[:]))}
 for _, fixture := range []workflowTestFixture{first, second} {
  for _, suffix := range []string{"", "-wal"} {
   body, err := os.ReadFile(fixture.databasePath + suffix)
   require.NoError(t, err)
   if suffix == "-wal" { require.Greater(t, len(body), 32, "populated WAL required for scan coverage") }
   for _, secret := range secretEncodings { require.False(t, bytes.Contains(body, secret), "synthetic Noise scalar persisted") }
   require.NotContains(t, string(body), "MESHTICULUM-PRIVATE-BACKEND-CANARY")
  }
 }
 t.Log("encrypted SQLite main/populated-WAL plaintext and Noise-scalar canary absence verified; fixture-only backend, no production activation")
}
